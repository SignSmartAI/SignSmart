            id="message"
            placeholder="Type a message here..."
        >

        <br>

        <button onclick="showMessage()">Show Message</button>
        <button onclick="waveCharacter()">Wave Hello!</button>

    <script>
        function showMessage() {
            const message = document.getElementById("message").value;
            const output = document.getElementById("output");

            if (message.trim() === "") {
                output.textContent = "Please type a message first!";
                return;
            }

            output.textContent = "Your message: " + message;

            const character = document.getElementById("character");
            character.classList.add("waving");

            setTimeout(function () {
                character.classList.remove("waving");
            }, 1500);
        }

        function waveCharacter() {
            const character = document.getElementById("character");

            character.classList.add("waving");

            document.getElementById("hands").textContent = "👋";
            document.getElementById("output").textContent =
                "Hello! Welcome to SignSmart AI!";

            setTimeout(function () {
                character.classList.remove("waving");
            }, 2000);
        }
    </script>
